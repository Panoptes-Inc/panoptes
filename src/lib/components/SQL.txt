-- Supabase SQL: waitlist + auth support (idempotent)

create table if not exists public.waitlist (
  id bigserial primary key,
  email text not null,
  auth_user_id uuid,
  created_at timestamptz not null default timezone('utc', now()),
  source text,
  page_path text
);

alter table public.waitlist
  add column if not exists website_url text,
  add column if not exists updated_at timestamptz not null default timezone('utc', now());

create index if not exists waitlist_created_at_idx on public.waitlist(created_at desc);
create index if not exists waitlist_email_lower_idx on public.waitlist ((lower(btrim(email))));

do $$
begin
  if not exists (
    select 1
    from pg_constraint
    where conname = 'waitlist_auth_user_id_fkey'
      and conrelid = 'public.waitlist'::regclass
  ) then
    alter table public.waitlist
      add constraint waitlist_auth_user_id_fkey
      foreign key (auth_user_id) references auth.users(id)
      on delete set null;
  end if;

  if not exists (
    select 1
    from pg_constraint
    where conname = 'waitlist_email_required_chk'
      and conrelid = 'public.waitlist'::regclass
  ) then
    alter table public.waitlist
      add constraint waitlist_email_required_chk
      check (length(btrim(email)) > 0);
  end if;

  if not exists (
    select 1
    from pg_constraint
    where conname = 'waitlist_source_len_chk'
      and conrelid = 'public.waitlist'::regclass
  ) then
    alter table public.waitlist
      add constraint waitlist_source_len_chk
      check (source is null or char_length(source) <= 64);
  end if;

  if not exists (
    select 1
    from pg_constraint
    where conname = 'waitlist_page_path_len_chk'
      and conrelid = 'public.waitlist'::regclass
  ) then
    alter table public.waitlist
      add constraint waitlist_page_path_len_chk
      check (page_path is null or char_length(page_path) <= 512);
  end if;

  if not exists (
    select 1
    from pg_constraint
    where conname = 'waitlist_website_url_len_chk'
      and conrelid = 'public.waitlist'::regclass
  ) then
    alter table public.waitlist
      add constraint waitlist_website_url_len_chk
      check (website_url is null or char_length(website_url) <= 1024);
  end if;
end $$;

create or replace function public.set_waitlist_updated_at()
returns trigger
language plpgsql
as $$
begin
  new.updated_at := timezone('utc', now());
  return new;
end;
$$;

drop trigger if exists trg_waitlist_updated_at on public.waitlist;
create trigger trg_waitlist_updated_at
before update on public.waitlist
for each row
execute function public.set_waitlist_updated_at();

alter table public.waitlist enable row level security;

drop policy if exists "waitlist deny direct access" on public.waitlist;
create policy "waitlist deny direct access"
on public.waitlist
for all
to anon, authenticated
using (false)
with check (false);

revoke all on table public.waitlist from public, anon, authenticated;

drop function if exists public.join_waitlist(text, text, text);

create or replace function public.join_waitlist(
  p_email text,
  p_source text default null,
  p_page_path text default null,
  p_website_url text default null
)
returns jsonb
language plpgsql
security definer
set search_path = public, pg_temp
as $$
declare
  v_email text := lower(btrim(coalesce(p_email, '')));
  v_source text := nullif(left(btrim(coalesce(p_source, '')), 64), '');
  v_page_path text := nullif(left(btrim(coalesce(p_page_path, '')), 512), '');
  v_website_url text := nullif(left(btrim(coalesce(p_website_url, '')), 1024), '');
  v_count int;
  v_limit int := 50;
begin
  if v_email = '' then
    return jsonb_build_object('ok', false, 'code', 'INVALID', 'message', 'email required');
  end if;

  if v_email !~* '^[A-Z0-9._%+\-]+@[A-Z0-9.\-]+\.[A-Z]{2,}$' then
    return jsonb_build_object('ok', false, 'code', 'INVALID', 'message', 'email format invalid');
  end if;

  if v_website_url is not null and v_website_url !~* '^https?://.+' then
    return jsonb_build_object('ok', false, 'code', 'INVALID', 'message', 'website_url format invalid');
  end if;

  -- Keep waitlist cap strict under concurrency.
  perform pg_advisory_xact_lock(hashtext('public.waitlist.capacity'));

  if exists (
    select 1
    from public.waitlist
    where lower(btrim(email)) = v_email
  ) then
    return jsonb_build_object('ok', true, 'code', 'ALREADY', 'message', 'already registered');
  end if;

  select count(*) into v_count from public.waitlist;
  if v_count >= v_limit then
    return jsonb_build_object('ok', false, 'code', 'FULL', 'message', 'waitlist is full');
  end if;

  insert into public.waitlist(email, source, page_path, website_url)
  values (v_email, v_source, v_page_path, v_website_url);

  return jsonb_build_object('ok', true, 'code', 'JOINED', 'message', 'joined');
exception
  when unique_violation then
    return jsonb_build_object('ok', true, 'code', 'ALREADY', 'message', 'already registered');
end;
$$;

revoke all on function public.join_waitlist(text, text, text, text) from public;
grant execute on function public.join_waitlist(text, text, text, text) to anon, authenticated;

create or replace function public.attach_waitlist_auth_user()
returns trigger
language plpgsql
security definer
set search_path = public, pg_temp
as $$
begin
  if new.email is null or length(btrim(new.email)) = 0 then
    return new;
  end if;

  update public.waitlist
  set auth_user_id = new.id
  where lower(btrim(email)) = lower(btrim(new.email))
    and (auth_user_id is null or auth_user_id = new.id);

  return new;
end;
$$;

drop trigger if exists trg_attach_waitlist_auth_user on auth.users;
create trigger trg_attach_waitlist_auth_user
after insert on auth.users
for each row
execute function public.attach_waitlist_auth_user();
