import type {
  AxisKey,
  Score100,
  StrategyStatusKey,
  Uuid,
} from "../core/types";

export interface StrategyImpactEstimate {
  target_axes: AxisKey[];
  score_lift_estimate: Record<AxisKey, number>;
}

export interface StrategyProposalCard {
  rank: 1 | 2 | 3;
  strategy_key: string;
  title: string;
  strategy_intent: string;
  rationale: string[];
  action_items: string[];
  expected_impact: StrategyImpactEstimate;
  adoptable: boolean;
}

export interface SelectedStrategyState {
  strategy_plan_id: Uuid;
  source_snapshot_id: Uuid;
  selected_strategy_key: string;
  selected_strategy_rank: 1 | 2 | 3;
  current_strategy_theme: string;
  strategy_status: StrategyStatusKey;
  previous_strategy_fit_index: Score100 | null;
  current_strategy_fit_index: Score100 | null;
  structural_risk_label: string;
  improvement_headroom_percent: number;
  selected_at: string;
  updated_at: string;
}

export interface PhaseTaskState {
  task_id: Uuid;
  phase_key: "plan_30" | "plan_60" | "plan_90";
  task_key: string;
  task_label: string;
  task_status: "not_started" | "in_progress" | "done" | "blocked";
  display_order: number;
  originated_from_snapshot_id: Uuid;
  carried_over_from_task_id: Uuid | null;
  archived: boolean;
  created_at: string;
  updated_at: string;
}

export interface PhaseExecutionState {
  phase_key: "plan_30" | "plan_60" | "plan_90";
  phase_label: string;
  phase_theme: string;
  phase_objective: string;
  progress_status: "not_started" | "in_progress" | "blocked" | "done";
  completed_task_count: number;
  total_task_count: number;
  assignee_status_label: string;
  next_action: string;
  tasks: PhaseTaskState[];
}

export interface StrategyExecutionState {
  strategy_plan_id: Uuid;
  project_id: Uuid;
  source_snapshot_id: Uuid;
  selected_strategy: SelectedStrategyState;
  plan_30: PhaseExecutionState;
  plan_60: PhaseExecutionState;
  plan_90: PhaseExecutionState;
  created_at: string;
  updated_at: string;
}

export interface PhaseTaskStateTransition {
  task_id: Uuid;
  from_status: "not_started" | "in_progress" | "done" | "blocked";
  to_status: "not_started" | "in_progress" | "done" | "blocked";
  changed_at: string;
  changed_by_user_id: Uuid | null;
  reason: string | null;
}

export interface ReanalysisTaskMatch {
  previous_task_id: Uuid;
  previous_task_label: string;
  new_task_key: string;
  new_task_label: string;
  match_type: "exact" | "semantic" | "none";
  carry_over_status: boolean;
}

export interface ReanalysisTaskMergeResult {
  strategy_plan_id: Uuid;
  previous_snapshot_id: Uuid;
  new_snapshot_id: Uuid;
  strategy_decision:
    | "strategy_keep"
    | "strategy_update_recommended"
    | "strategy_rework_required";
  carried_tasks: ReanalysisTaskMatch[];
  archived_task_ids: Uuid[];
  new_task_keys: string[];
}

export interface StrategyPlanPersistenceBundle {
  selected_strategy_state: SelectedStrategyState;
  execution_state: StrategyExecutionState;
}

import { z } from "zod";
import {
  axisKeySchema,
  phaseProgressKeySchema,
  score100Schema,
  strategyStatusKeySchema,
  uuidSchema,
} from "../core/schemas";

export const strategyImpactEstimateSchema = z.object({
  target_axes: z.array(axisKeySchema),
  score_lift_estimate: z.object({
    value_proposition: z.number(),
    information_hierarchy: z.number(),
    navigation_strength: z.number(),
    trust_proof: z.number(),
    differentiation: z.number(),
    decision_support: z.number(),
  }),
});

export const strategyProposalCardSchema = z.object({
  rank: z.union([z.literal(1), z.literal(2), z.literal(3)]),
  strategy_key: z.string().min(1),
  title: z.string().min(1),
  strategy_intent: z.string().min(1),
  rationale: z.array(z.string()),
  action_items: z.array(z.string()),
  expected_impact: strategyImpactEstimateSchema,
  adoptable: z.boolean(),
});

export const selectedStrategyStateSchema = z.object({
  strategy_plan_id: uuidSchema,
  source_snapshot_id: uuidSchema,
  selected_strategy_key: z.string().min(1),
  selected_strategy_rank: z.union([z.literal(1), z.literal(2), z.literal(3)]),
  current_strategy_theme: z.string().min(1),
  strategy_status: strategyStatusKeySchema,
  previous_strategy_fit_index: score100Schema.nullable(),
  current_strategy_fit_index: score100Schema.nullable(),
  structural_risk_label: z.string().min(1),
  improvement_headroom_percent: z.number().min(0).max(100),
  selected_at: z.string(),
  updated_at: z.string(),
});

export const phaseTaskStateSchema = z.object({
  task_id: uuidSchema,
  phase_key: z.enum(["plan_30", "plan_60", "plan_90"]),
  task_key: z.string().min(1),
  task_label: z.string().min(1),
  task_status: z.enum(["not_started", "in_progress", "done", "blocked"]),
  display_order: z.number().int().min(0),
  originated_from_snapshot_id: uuidSchema,
  carried_over_from_task_id: uuidSchema.nullable(),
  archived: z.boolean(),
  created_at: z.string(),
  updated_at: z.string(),
});

export const phaseExecutionStateSchema = z.object({
  phase_key: z.enum(["plan_30", "plan_60", "plan_90"]),
  phase_label: z.string().min(1),
  phase_theme: z.string().min(1),
  phase_objective: z.string().min(1),
  progress_status: phaseProgressKeySchema,
  completed_task_count: z.number().int().min(0),
  total_task_count: z.number().int().min(0),
  assignee_status_label: z.string().min(1),
  next_action: z.string().min(1),
  tasks: z.array(phaseTaskStateSchema),
});

export const strategyExecutionStateSchema = z.object({
  strategy_plan_id: uuidSchema,
  project_id: uuidSchema,
  source_snapshot_id: uuidSchema,
  selected_strategy: selectedStrategyStateSchema,
  plan_30: phaseExecutionStateSchema,
  plan_60: phaseExecutionStateSchema,
  plan_90: phaseExecutionStateSchema,
  created_at: z.string(),
  updated_at: z.string(),
});

export const phaseTaskStateTransitionSchema = z.object({
  task_id: uuidSchema,
  from_status: z.enum(["not_started", "in_progress", "done", "blocked"]),
  to_status: z.enum(["not_started", "in_progress", "done", "blocked"]),
  changed_at: z.string(),
  changed_by_user_id: uuidSchema.nullable(),
  reason: z.string().nullable(),
});

export const reanalysisTaskMatchSchema = z.object({
  previous_task_id: uuidSchema,
  previous_task_label: z.string().min(1),
  new_task_key: z.string().min(1),
  new_task_label: z.string().min(1),
  match_type: z.enum(["exact", "semantic", "none"]),
  carry_over_status: z.boolean(),
});

export const reanalysisTaskMergeResultSchema = z.object({
  strategy_plan_id: uuidSchema,
  previous_snapshot_id: uuidSchema,
  new_snapshot_id: uuidSchema,
  strategy_decision: z.enum([
    "strategy_keep",
    "strategy_update_recommended",
    "strategy_rework_required",
  ]),
  carried_tasks: z.array(reanalysisTaskMatchSchema),
  archived_task_ids: z.array(uuidSchema),
  new_task_keys: z.array(z.string()),
});

export const strategyPlanPersistenceBundleSchema = z.object({
  selected_strategy_state: selectedStrategyStateSchema,
  execution_state: strategyExecutionStateSchema,
});

3. 設計メモ
==================================================

- snapshot由来の推奨戦略と、現在の実行状態は分離して持つ
- 30/60/90 は phase summary ではなく task state を持つ実行管理構造とする
- 再分析時は task を全破棄しない
- task の引き継ぎ判定は exact / semantic / none を前提にする
- obsolete task は archived 扱いで残す
- 新規 task は not_started で追加する
