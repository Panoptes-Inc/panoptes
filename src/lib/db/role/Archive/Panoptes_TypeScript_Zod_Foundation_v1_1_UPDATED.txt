export const urlSchema = z.string().url();

export const score100Schema = z.number().min(0).max(100);

export const layerKeySchema = z.enum(LAYER_KEYS);
export const siteSizeKeySchema = z.enum(SITE_SIZE_KEYS);

export const projectStatusKeySchema = z.enum(PROJECT_STATUS_KEYS);
export const jobStatusKeySchema = z.enum(JOB_STATUS_KEYS);
export const analysisTypeKeySchema = z.enum(ANALYSIS_TYPE_KEYS);
export const analysisModeKeySchema = z.enum(ANALYSIS_MODE_KEYS);
export const narrativeToneKeySchema = z.enum(NARRATIVE_TONE_KEYS);
export const phaseKeySchema = z.enum(PHASE_KEYS);
export const phaseProgressKeySchema = z.enum(PHASE_PROGRESS_KEYS);
export const strategyStatusKeySchema = z.enum(STRATEGY_STATUS_KEYS);
export const marketPressureKeySchema = z.enum(MARKET_PRESSURE_KEYS);
export const confidenceLevelKeySchema = z.enum(CONFIDENCE_LEVEL_KEYS);
export const siteRoleKeySchema = z.enum(SITE_ROLE_KEYS);
export const pageTypeKeySchema = z.enum(PAGE_TYPE_KEYS);

export const axisScoreMapSchema = z.object({
  value_proposition: score100Schema.nullable(),
  information_hierarchy: score100Schema.nullable(),
  navigation_strength: score100Schema.nullable(),
  trust_proof: score100Schema.nullable(),
  differentiation: score100Schema.nullable(),
  decision_support: score100Schema.nullable(),
});

export const confidenceBandSchema = z.object({
  score: score01Schema,
  level: confidenceLevelKeySchema,
});

export const candidateScoreSchema = <T extends z.ZodTypeAny>(keySchema: T) =>
  z.object({
    key: keySchema,
    label: z.string().min(1),
    score: score01Schema,
  });

export const evidenceBundleSchema = z.object({
  keywords: z.array(z.string()),
  signals: z.array(z.string()),
  page_types: z.array(pageTypeKeySchema),
  notes: z.array(z.string()),
});

export const marketTypeDescriptorSchema = z.object({
  market_type_key: z.string().min(1),
  market_type_label: z.string().min(1),
  market_type_confidence: score01Schema,
  market_type_confidence_level: confidenceLevelKeySchema,
  market_type_tags: z.array(z.string()),
});

export const positionSummarySchema = z.object({
  position_label: z.string().min(1),
  market_pressure_level: marketPressureKeySchema,
  delta_vs_industry_total: z.number().nullable(),

  strengths: z.array(z.string()),

      structure_type_key: structureTypeKeySchema,
      structure_type_label: z.string().min(1),

  ),
